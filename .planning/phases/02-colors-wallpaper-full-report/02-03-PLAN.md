```python
def _rgb(t: tuple[int, int, int]) -> str:
    """Format an RGB triple as 'R,G,B' (KColorScheme convention; no alpha, no #)."""
    return f"{t[0]},{t[1]},{t[2]}"


def _twelve_keys(cs: ColorScheme) -> dict[str, str]:
    """The 12-key block emitted in every [Colors:*] section.

    Per RESEARCH §1 lines 134-148 and §2 line 287-293 — the same root values
    are inherited across Window/Button/View/Tooltip/Header/Complementary,
    with role-specific tweaks applied per-section by the caller.
    """
    return {
        "BackgroundNormal": _rgb(cs.background_normal),
        "BackgroundAlternate": _rgb(cs.background_alternate),
        "ForegroundNormal": _rgb(cs.foreground_normal),
        "ForegroundActive": _rgb(cs.foreground_active),
        "ForegroundInactive": _rgb(cs.foreground_inactive),
        "ForegroundLink": _rgb(cs.foreground_link),
        "ForegroundVisited": _rgb(cs.foreground_visited),
        "ForegroundPositive": _rgb(cs.foreground_positive),
        "ForegroundNeutral": _rgb(cs.foreground_neutral),
        "ForegroundNegative": _rgb(cs.foreground_negative),
        "DecorationFocus": _rgb(cs.decoration_focus),
        "DecorationHover": _rgb(cs.decoration_hover),
    }


def _selection_keys(cs: ColorScheme) -> dict[str, str]:
    """Selection variant: BackgroundNormal=DecorationFocus, ForegroundNormal=contrast.

    Per RESEARCH §2 line 290 — selected rows pop with the accent colour.
    """
    base = _twelve_keys(cs)
    base["BackgroundNormal"] = _rgb(cs.decoration_focus)
    # Choose contrasting foreground based on accent luminance
    r, g, b = cs.decoration_focus
    luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255.0
    base["ForegroundNormal"] = "255,255,255" if luminance < 0.5 else "0,0,0"
    return base
```

```python
def _can_roundtrip_subsection() -> bool:
    """Return True iff RawConfigParser can round-trip [Section][Sub] syntax.

    Pitfall E: KColorScheme uses [Colors:Header][Inactive] which is non-standard
    INI. We test once at module load whether RawConfigParser preserves the
    section name through write/read; if it doesn't, write_colors_ini appends
    that section as raw text instead.
    """
    cp = configparser.RawConfigParser()
    cp.optionxform = str  # type: ignore[method-assign]
    section_name = "Colors:Header][Inactive"  # configparser strips outer brackets
    try:
        cp.add_section(section_name)
        cp.set(section_name, "BackgroundNormal", "1,2,3")
        import io
        buf = io.StringIO()
        cp.write(buf, space_around_delimiters=False)
        text = buf.getvalue()
        return "[Colors:Header][Inactive]" in text
    except (configparser.Error, ValueError):
        return False


_SUBSECTION_ROUNDTRIPS = _can_roundtrip_subsection()
```

```python
def write_colors_ini(theme: Theme, out_path: Path) -> Path:
    """Write a Plasma 6 KColorScheme .colors file for the given Theme.

    Args:
        theme: Theme with a populated `color_scheme: ColorScheme` field. If
               theme.color_scheme is None, raises ValueError — the caller
               (Wave 3 install orchestrator) must ensure the field is set.
        out_path: absolute path to write to (typically
                  ~/.local/share/color-schemes/<name>.colors). Parent dir must
                  exist; this writer does not mkdir.

    Returns:
        out_path (the absolute path written), for the caller's manifest.

    Raises:
        ValueError: if theme.color_scheme is None.
    """
    if theme.color_scheme is None:
        raise ValueError(
            f"Theme {theme.name!r} has no color_scheme; cannot write .colors"
        )
    cs = theme.color_scheme
    cp = configparser.RawConfigParser()
    cp.optionxform = str  # type: ignore[method-assign]  # KDE keys are case-sensitive

    cp["General"] = {
        "ColorScheme": theme.name,
        "Name": theme.display_name,
        "shadeSortColumn": "true",
    }
    cp["KDE"] = {"contrast": "4"}
    cp["WM"] = {
        "activeBackground": _rgb(cs.wm_active_background),
        "activeForeground": _rgb(cs.wm_active_foreground),
        "activeBlend": _rgb(cs.wm_active_blend),
        "inactiveBackground": _rgb(cs.wm_inactive_background),
        "inactiveForeground": _rgb(cs.wm_inactive_foreground),
        "inactiveBlend": _rgb(cs.wm_inactive_blend),
    }

    base = _twelve_keys(cs)
    selection = _selection_keys(cs)

    # Most sections share the base 12-key block; Selection diverges per RESEARCH §2 line 290
    cp["Colors:Window"] = dict(base)
    cp["Colors:Button"] = dict(base)
    cp["Colors:View"] = dict(base)
    cp["Colors:Selection"] = selection
    cp["Colors:Tooltip"] = dict(base)
    cp["Colors:Header"] = dict(base)
    if _SUBSECTION_ROUNDTRIPS:
        cp["Colors:Header][Inactive"] = dict(base)
    cp["Colors:Complementary"] = dict(base)
    cp["ColorEffects:Disabled"] = dict(COLOR_EFFECTS_DISABLED)
    cp["ColorEffects:Inactive"] = dict(COLOR_EFFECTS_INACTIVE)

    with out_path.open("w", encoding="utf-8") as f:
        cp.write(f, space_around_delimiters=False)

    if not _SUBSECTION_ROUNDTRIPS:
        # Pitfall E fallback: append [Colors:Header][Inactive] as raw text
        lines = ["[Colors:Header][Inactive]"]
        for k, v in base.items():
            lines.append(f"{k}={v}")
        with out_path.open("a", encoding="utf-8") as f:
            f.write("\n" + "\n".join(lines) + "\n")

    return out_path
```

```python
JPEG_URI_PREFIX = "data:image/jpeg;base64,"


def embed_jpeg_b64(jpeg_bytes: bytes) -> str:
    """Wrap JPEG bytes in a base64 data URI with no newlines.

    Args:
        jpeg_bytes: Raw JPEG file bytes.

    Returns:
        A ``data:image/jpeg;base64,<b64>`` string suitable for use as an HTML
        ``<img src=...>`` attribute. Contains no whitespace or newlines.
    """
    b64 = base64.b64encode(jpeg_bytes).decode("ascii")
    return JPEG_URI_PREFIX + b64


def image_to_jpeg_b64_uri(img: Image.Image, quality: int = 80) -> str:
    """Save a Pillow Image as JPEG bytes at the given quality, embed as data URI.

    Per RESEARCH §6: HTML preview wallpaper thumbnails ship as JPEG (smaller
    payload than PNG for photographic content). quality=80 keeps a 320×240
    thumbnail under ~30 KB.

    Args:
        img: Any Pillow Image. Will be converted to RGB before JPEG save
             (JPEG does not support alpha; transparent pixels become black).
        quality: 1-95 JPEG quality. 80 is the project default for preview
                 thumbnails; lower values produce smaller URIs.

    Returns:
        A ``data:image/jpeg;base64,...`` data URI string.
    """
    buf = io.BytesIO()
    img.convert("RGB").save(buf, format="JPEG", quality=quality)
    return embed_jpeg_b64(buf.getvalue())
```

```python
def _safe_under(root: Path, candidate: Path) -> bool:
    """T-05-01 path-traversal guard: candidate must resolve inside root."""
    root_resolved = str(root.resolve())
    cand_resolved = str(candidate.resolve())
    return (
```

```python
def write_wallpaper_metadata(theme: Theme, out_dir: Path) -> Path:
    """Write metadata.json for a Plasma 6 Wallpaper/Images package.

    KPlugin.Id is derived from out_dir.name (NOT theme.name) so the package
    is self-consistent — Plasma rejects packages where the ID differs from
    the directory (RESEARCH Pitfall D).

    Args:
        theme: Theme — supplies display_name and author for KPlugin.Name/Authors.
        out_dir: directory the package lives in. Must already exist.

    Returns:
        Path to the written metadata.json.
    """
    data = {
        "KPlugin": {
            "Id": out_dir.name,  # MUST match dir name (Pitfall D)
            "Name": theme.display_name,
            "License": "unknown",  # E16 themes pre-date SPDX adoption
            "Version": "1.0",
            "Authors": [

                    "Email": "",
                }
            ],
        }
    }
    out_path = out_dir / "metadata.json"
    out_path.write_text(json.dumps(data, indent=4) + "\n", encoding="utf-8")
```

```python
def write_wallpaper_image(
    src: Path,
    contents_images_dir: Path,
    *,
    asset_root: Path | None = None,
) -> Path:
    """Copy or convert a wallpaper image into a Plasma wallpaper package layout.

    Reads the actual (w, h) from PIL after open (RESEARCH Pitfall C — never
    trust cfg-declared dimensions); writes to <contents_images_dir>/<W>x<H>.<ext>
    with lowercase 'x'.

    Args:
        src: source image path (PNG, JPG, GIF — others rejected with ValueError).
        contents_images_dir: target directory; will be mkdir(parents=True).
        asset_root: optional asset root for T-05-01 defence-in-depth check.
                    If supplied, src.resolve() must lie inside it; otherwise
                    raises ValueError.

    Returns:
        Path to the written image.

    Raises:
        ValueError: if src does not exist, is not a supported format, or
                    (when asset_root is supplied) escapes asset_root.
    """
    if not src.is_file():
        raise ValueError(f"wallpaper source does not exist: {src}")
    if asset_root is not None and not _safe_under(asset_root, src):
        raise ValueError(
            f"wallpaper path escapes asset_root: {src} not under {asset_root}"
        )

    suffix = src.suffix.lower()
    contents_images_dir.mkdir(parents=True, exist_ok=True)

    with Image.open(src) as img:
        w, h = img.size  # actual dimensions (Pitfall C)
        if suffix == ".gif":
            # Pitfall I: convert to PNG — Plasma's GIF support is unreliable
            out_path = contents_images_dir / f"{w}x{h}.png"
            img.convert("RGB").save(out_path, format="PNG")
        elif suffix in (".jpg", ".jpeg", ".png"):
            # Bytewise-faithful copy preserves source quality
            out_path = contents_images_dir / f"{w}x{h}{suffix}"
            shutil.copy2(src, out_path)
        else:
            raise ValueError(
                f"unsupported wallpaper format: {suffix} (supported: .png, .jpg, .gif)"
            )
    return out_path
```

```python
def write_wallpaper_package(theme: Theme, out_dir: Path) -> Path:
    """Write a complete Plasma 6 Wallpaper/Images package for a Theme.

    Creates:
        <out_dir>/metadata.json
        <out_dir>/contents/images/<W>x<H>.<ext>

    Args:
        theme: Theme with a populated `wallpaper: WallpaperSpec` field.
        out_dir: target directory (typically ~/.local/share/wallpapers/<name>).
                 The directory name MUST equal theme.name for KPlugin.Id
                 consistency.

    Returns:
        out_dir (the package root), for the caller's manifest.

    Raises:
        ValueError: if theme.wallpaper is None or wallpaper.image_path is None.
    """
    if theme.wallpaper is None or theme.wallpaper.image_path is None:
        raise ValueError(
            f"Theme {theme.name!r} has no wallpaper image; cannot write package"
        )
    out_dir.mkdir(parents=True, exist_ok=True)
    write_wallpaper_metadata(theme, out_dir)
    write_wallpaper_image(
        theme.wallpaper.image_path,
        out_dir / "contents" / "images",
        asset_root=theme.asset_root,
    )
    return out_dir
```
