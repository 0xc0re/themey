    ```python
    """Single-theme conversion pipeline.

    Composes the four stages: ingest → analyze → generate → install + report
    + preview. The asset_root from archive.extract is valid ONLY inside the
    `with` block, so all generate-stage work happens inside it.
    """
    from __future__ import annotations

    import logging
    import shutil
    import tempfile
    from dataclasses import dataclass
    from pathlib import Path

    from . import paths, install
    from .analyze.build_theme import build_theme
    from .etheme.archive import extract
    from .etheme.parse import parse_tree
    from .generate.aurorae import write as write_aurorae
    from .preview import render as render_preview
    from .report import write as write_report
    from .slug import slugify

    log = logging.getLogger(__name__)


    @dataclass(frozen=True)
    class ConvertResult:
        theme_name: str
        installed_dir: Path
        preview_path: Path
        report_path: Path
        notes_count: int


    def convert(etheme_path: Path, *, scale: int = 2) -> ConvertResult:
        """Convert one .etheme to an installed Aurorae theme + preview + report."""
        if scale not in (1, 2, 3):
            raise ValueError(f"scale must be 1, 2, or 3 (got {scale})")
        # Theme name comes from the archive filename, never from cfg content.
        theme_name = slugify(etheme_path.stem)
        log.info("converting %s as %s (scale=%d)", etheme_path, theme_name, scale)

        # Stage outputs under XDG_DATA_HOME so os.replace can rename atomically
        # into ~/.local/share/aurorae/themes/<name>/ on the same filesystem.
        staging_root = paths.themey_previews().parent / "staging"
        staging_root.mkdir(parents=True, exist_ok=True)

        with extract(etheme_path) as raw:
            log.debug("extracted to %s", raw.asset_root)
            ast_nodes = parse_tree(raw.asset_root)
            log.debug("parsed %d top-level AST nodes", len(ast_nodes))
            theme = build_theme(
                raw.asset_root, ast_nodes,
                name=theme_name,
                display_name=theme_name,
                scale=scale,
            )
            log.info("theme: parts=%d iclasses=%d notes=%d skipped=%d",
                     len(theme.border.parts), len(theme.iclasses),
                     len(theme.notes), len(theme.skipped_borders))

            # Stage the Aurorae output to a tmp dir under staging_root
            with tempfile.TemporaryDirectory(
                prefix=f"{theme_name}-", dir=str(staging_root),
            ) as stage_str:
                stage = Path(stage_str)
                # Generators write into stage/<theme_name>/
                stage_theme_dir = stage / theme_name
                write_aurorae(theme, stage_theme_dir)
                log.debug("wrote Aurorae output to %s", stage_theme_dir)
                # Atomic install
                installed = install.deploy(theme_name, stage_theme_dir)
                log.info("installed to %s", installed)

            # Report and preview live under XDG_DATA_HOME/themey/previews/
            previews = paths.themey_previews()
            previews.mkdir(parents=True, exist_ok=True)
            report_path = write_report(theme, previews / f"{theme_name}.report.txt")
            preview_path = render_preview(theme, previews / f"{theme_name}.html")

        return ConvertResult(
            theme_name=theme_name,
            installed_dir=installed,
            preview_path=preview_path,
            report_path=report_path,
            notes_count=len(theme.notes),
        )
    ```

    A few implementation details to call out for the executor:
    - The `with tempfile.TemporaryDirectory(... dir=str(staging_root))` ensures the staged output is on the same filesystem as the install target, so `os.replace` is atomic.
    - install.deploy renames `stage_theme_dir` (a path inside the TemporaryDirectory) into the final position. The TemporaryDirectory cleanup will then be a no-op for that subdir (since it's been renamed away).
    - **Lifecycle invariant:** `raw.asset_root` is only valid while the `with extract(...)` block is open. `build_theme` and `write_aurorae` both run INSIDE that block. After the block exits, the tmp tree is gone. Report and preview run AFTER the block exits — they only consume `theme: Theme` data, NOT `raw.asset_root`. Verify this in the implementation.

    Wait — `build_theme` stores `asset_root` on `Theme.asset_root`. After `with extract(...)` exits, `theme.asset_root` becomes a stale reference. Report and preview don't use `theme.asset_root` (verified: report.py and preview.py only read theme.name, theme.notes, theme.palette, etc.). So the lifecycle is OK as long as report and preview never read theme.asset_root. **Document this in pipeline.py docstring** so future code doesn't accidentally introduce a bug.

    ```python
    """themey CLI entry point.

    Single-theme form (Phase 1; CLI-01):
        themey <theme.etheme>

    Flags:
        --scale=N   1, 2 (default), or 3
        -v / -vv    increase verbosity (DEBUG, default INFO)
        -q          quiet (WARNING+ only)

    Batch form (themey --all <dir>) is Phase 4 and intentionally not exposed
    here. The roadmap maps CLI-01 entirely to Phase 1 because the single-theme
    form ships first.
    """
    from __future__ import annotations

    import logging
    import sys
    from pathlib import Path

    import typer

    from . import external, log
    from .pipeline import ConvertResult, convert

    app = typer.Typer(no_args_is_help=True, add_completion=False)


    @app.command()
    def convert_cmd(
        theme: Path = typer.Argument(
            ..., exists=True, dir_okay=False, readable=True,
            help="Path to a .etheme archive",
        ),
        scale: int = typer.Option(
            2, "--scale", min=1, max=3,
            help="Border/image upscale factor (1/2/3; default 2)",
        ),
        verbose: int = typer.Option(0, "-v", "--verbose", count=True,
                                     help="Increase verbosity"),
        quiet: bool = typer.Option(False, "-q", "--quiet",
                                    help="Suppress info messages"),
    ) -> None:
        """Convert one .etheme to a Plasma 6 Aurorae window decoration."""
        log.setup_logging(verbose=verbose, quiet=quiet)
        try:
            result = convert(theme, scale=scale)
        except Exception as exc:  # noqa: BLE001 -- top-level CLI handler
            logging.getLogger(__name__).error("conversion failed: %s", exc)
            raise typer.Exit(code=1) from exc

        typer.echo(f"Installed: {result.installed_dir}")
        typer.echo(f"Preview:   {result.preview_path}")
        typer.echo(f"Report:    {result.report_path}")
        typer.echo(f"Apply via System Settings - Window Decorations - "
                   f"{result.theme_name}")

        # Auto-open preview unless headless / SSH
        opened = external.open_preview_unless_headless(result.preview_path)
        if not opened:
            typer.echo(f"(Open the preview manually: file://{result.preview_path})")


    if __name__ == "__main__":
        app()
    ```

| .etheme content → entire pipeline | safe_extract is the only entry point for archive content; everything downstream trusts the validated tmpdir. |
| Pipeline result → user's filesystem | install.deploy is the only writer to ~/.local/share/aurorae/themes/. atomic + rollback. |
