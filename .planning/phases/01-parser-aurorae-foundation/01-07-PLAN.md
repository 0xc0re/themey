    ```python
    """Hand-rolled .desktop INI writer.

    DO NOT use configparser — localization keys like `Name[de]=Foo` look like
    section headers to configparser's regex.
    """
    from __future__ import annotations

    from pathlib import Path


    def write_desktop(path: Path, sections: dict[str, dict[str, str]]) -> None:
        with open(path, "w", encoding="utf-8") as f:
            first = True
            for section, entries in sections.items():
                if not first:
                    f.write("\n")
                f.write(f"[{section}]\n")
                for k, v in entries.items():
                    f.write(f"{k}={v}\n")
                first = False
    ```

    ```python
    """Aurorae <name>rc INI writer.

    RawConfigParser preserves '%' (some KDE color values), and optionxform=str
    preserves KDE's case-sensitive keys (LeftButtons, BackgroundNormal, etc.).

    Layout numerical mapping is documented as constants — Open Question A2 in
    01-RESEARCH.md flags these as visual-gate-tunable.
    """
    from __future__ import annotations

    import configparser
    from pathlib import Path

    from themey.ir import Theme


    def _format_color_rgba(rgb: tuple[int, int, int]) -> str:
        return f"{rgb[0]},{rgb[1]},{rgb[2]},255"


    def _layout_values(theme: Theme) -> dict[str, str]:
        s = theme.scale
        bl = theme.border.border_size_left * s
        br = theme.border.border_size_right * s
        bt = theme.border.border_size_top * s
        bb = theme.border.border_size_bottom * s
        return {
            "BorderLeft":         str(bl),
            "BorderRight":        str(br),
            "BorderBottom":       str(bb),
            "BorderTop":          str(bt),
            "TitleEdgeTop":       str(2 * s),
            "TitleEdgeBottom":    str(2 * s),
            "TitleEdgeLeft":      str(4 * s),
            "TitleEdgeRight":     str(4 * s),
            "TitleBorderLeft":    str(2 * s),
            "TitleBorderRight":   str(2 * s),
            "TitleHeight":        str(max(15, bt - 4 * s)),
            "ButtonWidth":        str(12 * s),
            "ButtonHeight":       str(12 * s),
            "ButtonSpacing":      str(4 * s),
            "ButtonMarginTop":    str(2 * s),
            "ButtonMarginLeft":   str(3 * s),
            "ExplicitButtonSpacer": "0",
            "PaddingTop":         str(bt * 2),  # FrameSvg shadow padding
            "PaddingBottom":      str(bb * 2),
            "PaddingLeft":        str(bl * 2),
            "PaddingRight":       str(br * 2),
        }


    class _CaseSensitiveRawConfigParser(configparser.RawConfigParser):
        # Subclass to preserve KDE-required key case (LeftButtons,
        # BackgroundNormal, etc.) WITHOUT triggering pyright basic's
        # "cannot assign method to instance" complaint that the bare
        # `cp.optionxform = str` form raises.
        optionxform = staticmethod(str)


    def write_aurorae_rc(theme: Theme, out_dir: Path) -> Path:
        cp = _CaseSensitiveRawConfigParser()

        cp["General"] = {
            "ActiveTextColor":   _format_color_rgba(theme.palette.text_active),
            "InactiveTextColor": _format_color_rgba(theme.palette.text_inactive),
            "TitleAlignment": "Center",
            "TitleVerticalAlignment": "Center",
            "UseTextShadow": "true",
            "ActiveTextShadowColor": "0,0,0,255",
            "InactiveTextShadowColor": "0,0,0,255",
            "TextShadowOffsetX": "0",
            "TextShadowOffsetY": "1",
            "LeftButtons":  theme.left_buttons,
            "RightButtons": theme.right_buttons,
            "Shadow":       "true",
            "Animation":    "1",
        }
        cp["Layout"] = _layout_values(theme)

        out_path = out_dir / f"{theme.name}rc"
        with open(out_path, "w", encoding="utf-8") as f:
            cp.write(f, space_around_delimiters=False)
        return out_path
    ```

    ```python
    """decoration.svg writer with 18 FrameSvg IDs + hint margins.

    Pitfall 5: 18 IDs (`decoration-{topleft,top,topright,left,center,right,
    bottomleft,bottom,bottomright}` × {active, inactive}) verbatim — Aurorae
    matches by literal id with `decoration-` prefix.

    Pitfall 6: every <image> uses `data:image/png;base64,...` href and
    preserveAspectRatio="none". Relative paths fail after install relocation.

    Maximized variants are NOT emitted (Edna ships without them and works;
    01-RESEARCH.md Pitfall 5 closing note + assumption A8).
    """
    from __future__ import annotations

    import io
    import xml.etree.ElementTree as ET
    from pathlib import Path

    from PIL import Image

    from themey.images.embed import embed_png_b64
    from themey.images.upscale import upscale_nearest
    from themey.ir import IClassSpec, Theme

    SVG_NS = "http://www.w3.org/2000/svg"
    XLINK_NS = "http://www.w3.org/1999/xlink"

    SIDES: tuple[str, ...] = (
        "topleft", "top", "topright",
        "left", "center", "right",
        "bottomleft", "bottom", "bottomright",
    )


    def _png_bytes(p: Path | None, scale: int) -> bytes | None:
        if p is None or not p.is_file():
            return None
        with Image.open(p) as src:
            src = src.convert("RGBA")
            scaled = upscale_nearest(src, scale)
            buf = io.BytesIO()
            scaled.save(buf, format="PNG")
            return buf.getvalue()


    def _select_titlebar_iclass(theme: Theme) -> IClassSpec | None:
        # Prefer TITLE_BAR_HORIZONTAL; fall back to first iclass that has __NORMAL
        if "TITLE_BAR_HORIZONTAL" in theme.iclasses:
            return theme.iclasses["TITLE_BAR_HORIZONTAL"]
        for ic in theme.iclasses.values():
            if ic.normal is not None:
                return ic
        return None


    def _compute_region_bbox(side: str, w: int, h: int,
                              edge_l: int, edge_r: int,
                              edge_t: int, edge_b: int) -> tuple[int, int, int, int]:
        """Return (x, y, width, height) for a 9-patch region inside the canvas."""
        cw = max(0, w - edge_l - edge_r)
        ch = max(0, h - edge_t - edge_b)
        return {
            "topleft":      (0,            0,            edge_l, edge_t),
            "top":          (edge_l,       0,            cw,     edge_t),
            "topright":     (edge_l + cw,  0,            edge_r, edge_t),
            "left":         (0,            edge_t,       edge_l, ch),
            "center":       (edge_l,       edge_t,       cw,     ch),
            "right":        (edge_l + cw,  edge_t,       edge_r, ch),
            "bottomleft":   (0,            edge_t + ch,  edge_l, edge_b),
            "bottom":       (edge_l,       edge_t + ch,  cw,     edge_b),
            "bottomright":  (edge_l + cw,  edge_t + ch,  edge_r, edge_b),
        }[side]


    def write_decoration_svg(theme: Theme, out_dir: Path) -> Path:
        ET.register_namespace("", SVG_NS)
        ET.register_namespace("xlink", XLINK_NS)

        ic = _select_titlebar_iclass(theme)
        scale = theme.scale

        # Pick PNG bytes for active and inactive states
        active_bytes = _png_bytes(ic.normal_active if ic else None, scale)
        if active_bytes is None and ic is not None:
            active_bytes = _png_bytes(ic.normal, scale)
        inactive_bytes = _png_bytes(ic.normal if ic else None, scale)
        if inactive_bytes is None:
            inactive_bytes = active_bytes

        # If neither resolved, generate a 1x1 transparent PNG so the SVG still
        # has all 18 IDs satisfied (degraded mode; surfaced via Theme.notes
        # by Plan 05).
        if active_bytes is None:
            with Image.new("RGBA", (1, 1), (0, 0, 0, 0)) as blank:
                buf = io.BytesIO()
                blank.save(buf, format="PNG")
                active_bytes = buf.getvalue()
        if inactive_bytes is None:
            inactive_bytes = active_bytes

        # Canvas size — derive from edge_scaling × scale, with a sensible body.
        if ic is not None:
            el, er, et, eb = ic.edge_scaling
        else:
            el, er, et, eb = (4, 4, 18, 4)
        el *= scale; er *= scale; et *= scale; eb *= scale

        # Pick a body region that's visually meaningful (the body is what
        # FrameSvg stretches; we just need IDs at correct positions).
        body_w = max(64, el + er + 32)
        body_h = max(64, et + eb + 32)

        svg = ET.Element(f"{{{SVG_NS}}}svg", {
            "width": str(body_w),
            "height": str(body_h),
            "version": "1.1",
        })

        for prefix, png_bytes in (("decoration", active_bytes),
                                   ("decoration-inactive", inactive_bytes)):
            href = embed_png_b64(png_bytes)
            for side in SIDES:
                g = ET.SubElement(svg, f"{{{SVG_NS}}}g", {"id": f"{prefix}-{side}"})
                x, y, rw, rh = _compute_region_bbox(side, body_w, body_h,
                                                     el, er, et, eb)
                ET.SubElement(g, f"{{{SVG_NS}}}image", {
                    f"{{{XLINK_NS}}}href": href,
                    "x": str(x), "y": str(y),
                    "width": str(max(1, rw)), "height": str(max(1, rh)),
                    "preserveAspectRatio": "none",
                })

        # FrameSvg hint margins — 1×<edge> invisible rects telling FrameSvg the
        # border thickness in each direction (verified per Edna decoration.svg).
        for hint_id, dim in (
            ("hint-top-margin", et),
            ("hint-bottom-margin", eb),
            ("hint-left-margin", el),
            ("hint-right-margin", er),
        ):
            ET.SubElement(svg, f"{{{SVG_NS}}}rect", {
                "id": hint_id, "x": "0", "y": "0",
                "width":  str(max(1, dim)),
                "height": str(max(1, dim)),
                "style":  "opacity:0",
            })

        out_path = out_dir / "decoration.svg"
        ET.ElementTree(svg).write(out_path, xml_declaration=True,
                                   encoding="utf-8")

        # Validate the 18 required IDs are present
        from themey.generate.aurorae import REQUIRED_FRAMESVG_IDS
        root = ET.parse(out_path).getroot()
        present = {e.get("id") for e in root.iter() if e.get("id")}
        missing = set(REQUIRED_FRAMESVG_IDS) - present
        if missing:
            raise AssertionError(f"FrameSvg IDs missing in decoration.svg: {missing}")

        return out_path
    ```

    ```python
    """Per-button SVG writer.

    Aurorae loads close.svg / maximize.svg / restore.svg / minimize.svg /
    shade.svg / alldesktops.svg / keepabove.svg / keepbelow.svg from the
    theme dir based on the button codes in <name>rc's LeftButtons/RightButtons.
    Each SVG should contain elements with ids matching `*-hover` and
    `*-pressed` so Aurorae renders feedback states.
    """
    from __future__ import annotations

    import io
    import xml.etree.ElementTree as ET
    from pathlib import Path

    from PIL import Image

    from themey.images.embed import embed_png_b64
    from themey.images.upscale import upscale_nearest
    from themey.ir import Theme

    SVG_NS = "http://www.w3.org/2000/svg"
    XLINK_NS = "http://www.w3.org/1999/xlink"

    # Aurorae button code → list of files to write (some codes write multiple)
    BUTTON_CODE_TO_FILES: dict[str, tuple[str, ...]] = {
        "X": ("close.svg",),
        "A": ("maximize.svg", "restore.svg"),
        "I": ("minimize.svg",),
        "S": ("alldesktops.svg",),
        "L": ("shade.svg",),
        "F": ("keepabove.svg",),
        "B": ("keepbelow.svg",),
    }

    # Map Aurorae code back to E16 iclass-name pattern, for finding the source
    # PNG to embed. The codes match `BUTTON_*` by name in the theme.iclasses.
    CODE_TO_ICLASS_PATTERNS: dict[str, tuple[str, ...]] = {
        "X": ("BUTTON_CLOSE", "BUTTON_KILL"),
        "A": ("BUTTON_MAXIMIZE", "BUTTON_MAX"),
        "I": ("BUTTON_ICONIFY", "BUTTON_MINIMIZE"),
        "S": ("BUTTON_STICK",),
        "L": ("BUTTON_SHADE",),
        "F": ("BUTTON_KEEP_ABOVE",),
        "B": ("BUTTON_KEEP_BELOW",),
    }


    def _png_bytes(p: "Path | None", scale: int) -> bytes | None:
        if p is None or not Path(p).is_file():
            return None
        with Image.open(p) as src:
            src = src.convert("RGBA")
            scaled = upscale_nearest(src, scale)
            buf = io.BytesIO()
            scaled.save(buf, format="PNG")
            return buf.getvalue()


    def _find_iclass_for_code(theme: Theme, code: str):
        for pattern in CODE_TO_ICLASS_PATTERNS.get(code, ()):
            for ic_name, ic in theme.iclasses.items():
                if pattern in ic_name.upper():
                    return ic
        return None


    def _build_button_svg(theme: Theme, code: str, base_id: str) -> ET.Element:
        ET.register_namespace("", SVG_NS)
        ET.register_namespace("xlink", XLINK_NS)
        ic = _find_iclass_for_code(theme, code)
        scale = theme.scale

        normal = _png_bytes(ic.normal_active if ic else None, scale) \
                  or _png_bytes(ic.normal if ic else None, scale)
        hover = _png_bytes(ic.hilited_active if ic else None, scale) \
                 or _png_bytes(ic.hilited if ic else None, scale) \
                 or normal
        pressed = _png_bytes(ic.clicked_active if ic else None, scale) \
                   or _png_bytes(ic.clicked if ic else None, scale) \
                   or normal

        # Fallback: 16x16 transparent
        if normal is None:
            with Image.new("RGBA", (16, 16), (0, 0, 0, 0)) as blank:
                buf = io.BytesIO()
                blank.save(buf, format="PNG")
                normal = hover = pressed = buf.getvalue()

        size = 24 * scale
        svg = ET.Element(f"{{{SVG_NS}}}svg", {
            "width":   str(size),
            "height":  str(size),
            "version": "1.1",
        })
        # default state (this id is what Aurorae uses for the inactive/idle button)
        for sub_id, png_bytes in ((base_id, normal),
                                    (f"{base_id}-hover", hover),
                                    (f"{base_id}-pressed", pressed)):
            g = ET.SubElement(svg, f"{{{SVG_NS}}}g", {"id": sub_id})
            ET.SubElement(g, f"{{{SVG_NS}}}image", {
                f"{{{XLINK_NS}}}href": embed_png_b64(png_bytes),
                "x": "0", "y": "0",
                "width": str(size), "height": str(size),
                "preserveAspectRatio": "none",
            })
        return svg


    def write_button_svgs(theme: Theme, out_dir: Path) -> list[Path]:
        codes = set(theme.left_buttons) | set(theme.right_buttons)
        # Also accept codes inferred from button_codes dict (defensive)
        codes |= set(theme.button_codes.values())
        written: list[Path] = []
        for code in codes:
            for fname in BUTTON_CODE_TO_FILES.get(code, ()):
                base_id = fname.removesuffix(".svg")  # e.g. "close"
                svg = _build_button_svg(theme, code, base_id)
                out_path = out_dir / fname
                ET.ElementTree(svg).write(
                    out_path, xml_declaration=True, encoding="utf-8"
                )
                written.append(out_path)
        return written
    ```

|----------|-------------|
| Theme.iclasses Path values → Pillow.Image.open | Asset paths come from cfg + asset_root; safe_extract has validated all paths inside asset_root, but Pillow may still receive a path to a corrupt PNG. |
| out_dir → caller (Plan 08 install.py) | Generators write into a tmp out_dir; trust that Plan 08 atomically moves it into ~/.local/share/aurorae/themes/. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-01 | Tampering | corrupt PNG asset crashes Pillow.Image.open with cryptic error | mitigate | `_png_bytes` checks `p.is_file()` before opening; if Pillow raises during convert("RGBA") or save("PNG"), let it propagate (Plan 09 catches and emits clear message). Phase 1 fail-loudly per anti-pattern rule. |
| T-07-02 | Tampering | malicious PNG with embedded ICC profile or huge alpha layer balloons memory | accept | Pillow has its own bomb mitigations; safe_extract's 8 MB cap bounds source size. Plan 07 may also enforce a post-write 2 MB warning per Pitfall 12. |
| T-07-03 | Tampering | metadata.json injects shell-special characters into theme.display_name → preview HTML XSS | mitigate | Theme.display_name comes from theme.name (slug from filename); never from cfg content. Plan 08 preview.py also `html.escape()`s any rendered field. |
| T-07-04 | Tampering | XML injection via theme name with `<` or `>` characters | mitigate | xml.etree's `ET.SubElement` and `Element` properly escape attribute values; theme.name is slugified before reaching this layer (Plan 09 sanitizes). |
| T-07-05 | DoS | decoration.svg larger than ~2 MB triggers FrameSvg cache misbehavior (Pitfall 12) | mitigate | Plan 08 install.py checks output size and appends a warning to report.txt when > 2 MB (delegated to Plan 08 since report.py is there).
