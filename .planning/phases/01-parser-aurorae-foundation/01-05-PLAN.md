    Implementation strategy: each tclass / iclass / border block has a "current state context" set by E16's flat `__NORMAL` / `__NORMAL_ACTIVE` / `__HILITED_ACTIVE` / `__CLICKED_ACTIVE` / `__HILITED` / `__CLICKED` / `__NORMAL_STICKY` / `__NORMAL_ACTIVE_STICKY` keywords. Each KeyVal that follows attaches to the current state. **Implementation rule:** walk the Block.children in order, maintaining a `current_state: str` variable that updates when one of these keywords is seen as a KeyVal with no values (or zero values). KeyVals like `__FORGROUND_COLOR R G B` or `__NORMAL "path.png"` (string image path) attach to the current state.

    Wait — that's the E16 way for some blocks (TCLASS) but for ICLASS the convention is different. Per WILBS reference and 01-RESEARCH.md, ICLASSes use a flat `__NORMAL "path.png"` form where the state name takes a value (the path) directly:

    ```
    __ICLASS BUTTON_CLOSE
    __BGN
    __EDGE_SCALING 0 0 0 0
    __NORMAL "btn_close.png"
    __NORMAL_ACTIVE "btn_close.png"
    __HILITED_ACTIVE "btn_close_hover.png"
    __END
    ```

    But TCLASS uses the state-context form:
    ```
    __TCLASS TEXT1
    __BGN
    __NORMAL
    __FORGROUND_COLOR 200 200 150
    __NORMAL_ACTIVE
    __FORGROUND_COLOR 255 255 200
    __END
    ```

    **Both forms must be supported.** For ICLASS: `__NORMAL <path>` is a KeyVal with one value. For TCLASS: `__NORMAL` is a KeyVal with zero values that sets the current state context for the next `__FORGROUND_COLOR R G B` KeyVal. The implementer handles both per-block-type.

    ```python
    """AST __BORDER block → BorderSpec + ButtonPart extraction.

    DEFAULT-only selection per CLAUDE.md / WILBS-REFERENCE.md Section 6.
    """
    from __future__ import annotations

    from themey.etheme.ast import AstNode, Block, KeyVal
    from themey.ir import BorderSpec, ButtonPart


    def select_default_border(borders: list[Block]) -> Block | None:
        """Pick the DEFAULT border. Selection rule:
        1. Block with head_values[0] == 'DEFAULT'
        2. First block with at least one __BORDER_PART child
        3. None
        """
        if not borders:
            return None
        for b in borders:
            if b.head_values and b.head_values[0] == "DEFAULT":
                return b
        for b in borders:
            for c in b.children:
                if isinstance(c, Block) and c.keyword == "__BORDER_PART":
                    return b
        return None


    def extract_button_parts(border: Block) -> tuple[ButtonPart, ...]:
        parts: list[ButtonPart] = []
        for child in border.children:
            if not (isinstance(child, Block) and child.keyword == "__BORDER_PART"):
                continue
            iclass = ""
            aclass: str | None = None  # null sentinel
            coords = {
                "tl_x_pct": 0, "tl_x_abs": 0, "tl_y_pct": 0, "tl_y_abs": 0,
                "br_x_pct": 0, "br_x_abs": 0, "br_y_pct": 0, "br_y_abs": 0,
            }
            for kv in child.children:
                if not isinstance(kv, KeyVal):
                    continue
                k = kv.keyword
                if k == "__ICLASS" and kv.values:
                    iclass = str(kv.values[0])
                elif k == "__ACLASS" and kv.values:
                    aclass = str(kv.values[0])
                elif k == "__TOPLEFT_X_PERCENTAGE" and kv.values:
                    coords["tl_x_pct"] = int(kv.values[0])
                elif k == "__TOPLEFT_X_ABSOLUTE" and kv.values:
                    coords["tl_x_abs"] = int(kv.values[0])
                elif k == "__TOPLEFT_Y_PERCENTAGE" and kv.values:
                    coords["tl_y_pct"] = int(kv.values[0])
                elif k == "__TOPLEFT_Y_ABSOLUTE" and kv.values:
                    coords["tl_y_abs"] = int(kv.values[0])
                elif k == "__BOTTOMRIGHT_X_PERCENTAGE" and kv.values:
                    coords["br_x_pct"] = int(kv.values[0])
                elif k == "__BOTTOMRIGHT_X_ABSOLUTE" and kv.values:
                    coords["br_x_abs"] = int(kv.values[0])
                elif k == "__BOTTOMRIGHT_Y_PERCENTAGE" and kv.values:
                    coords["br_y_pct"] = int(kv.values[0])
                elif k == "__BOTTOMRIGHT_Y_ABSOLUTE" and kv.values:
                    coords["br_y_abs"] = int(kv.values[0])
            parts.append(ButtonPart(iclass_name=iclass, aclass=aclass, **coords))
        return tuple(parts)


    def build_border(border_block: Block) -> BorderSpec:
        sizes = {"left": 0, "right": 0, "top": 0, "bottom": 0}
        for kv in border_block.children:
            if not isinstance(kv, KeyVal) or not kv.values:
                continue
            if kv.keyword == "__BORDER_SIZE_LEFT":   sizes["left"] = int(kv.values[0])
            elif kv.keyword == "__BORDER_SIZE_RIGHT":  sizes["right"] = int(kv.values[0])
            elif kv.keyword == "__BORDER_SIZE_TOP":    sizes["top"] = int(kv.values[0])
            elif kv.keyword == "__BORDER_SIZE_BOTTOM": sizes["bottom"] = int(kv.values[0])
        name = border_block.head_values[0] if border_block.head_values else "DEFAULT"
        return BorderSpec(
            name=str(name),
            border_size_left=sizes["left"],
            border_size_right=sizes["right"],
            border_size_top=sizes["top"],
            border_size_bottom=sizes["bottom"],
            parts=extract_button_parts(border_block),
        )
    ```

    ```python
    """AST __ICLASS blocks → IClassSpec dict + raw state map for AURORAE-04 collapse.

    EDGE_SCALING order is L R T B (verified in iclass.c sscanf order; see Pitfall 1).
    """
    from __future__ import annotations

    from pathlib import Path

    from themey.etheme.ast import AstNode, Block, KeyVal
    from themey.ir import IClassSpec

    # State keywords whose presence as a KeyVal with one string value sets the
    # iclass image for that state. (E16 ICLASS uses inline form.)
    ICLASS_STATE_KEYS: frozenset[str] = frozenset({
        "__NORMAL", "__NORMAL_ACTIVE",
        "__HILITED", "__HILITED_ACTIVE",
        "__CLICKED", "__CLICKED_ACTIVE",
        "__NORMAL_STICKY", "__NORMAL_ACTIVE_STICKY",
        "__CLICKED_STICKY", "__CLICKED_ACTIVE_STICKY",
        "__HILITED_STICKY", "__HILITED_ACTIVE_STICKY",
        "__NORMAL_ACTIVE_CLICKED", "__DISABLED",
    })


    def build_iclasses(
        iclass_blocks: list[Block],
        asset_root: Path,
    ) -> tuple[dict[str, IClassSpec], dict[str, dict[str, Path | None]]]:
        """Returns (typed_dict, raw_state_dict).

        typed_dict: name → IClassSpec (subset of states surfaced on the dataclass)
        raw_state_dict: name → {state_keyword: Path | None}  (full state map for
                        AURORAE-04 collapse — includes sticky/disabled/etc.)
        """
        typed: dict[str, IClassSpec] = {}
        raw: dict[str, dict[str, Path | None]] = {}
        for block in iclass_blocks:
            if not block.head_values:
                continue
            name = str(block.head_values[0])
            edge = (0, 0, 0, 0)  # left, right, top, bottom
            states: dict[str, Path | None] = {}
            for kv in block.children:
                if not isinstance(kv, KeyVal):
                    continue
                if kv.keyword == "__EDGE_SCALING" and len(kv.values) >= 4:
                    edge = (int(kv.values[0]), int(kv.values[1]),
                            int(kv.values[2]), int(kv.values[3]))
                elif kv.keyword in ICLASS_STATE_KEYS and kv.values:
                    p = str(kv.values[0])
                    # Resolve relative to asset_root; do NOT resolve symlinks
                    # (already done by safe_extract).
                    # POLICY (resolved in Plan 01-05 revision iteration 1):
                    # store the resolved Path unconditionally; never None
                    # for missing files. Callers check `.is_file()` themselves.
                    # build_theme appends a missing-asset note when the file
                    # does not exist on disk.
                    states[kv.keyword] = (asset_root / p).resolve()
            raw[name] = states
            typed[name] = IClassSpec(
                name=name,
                edge_scaling=edge,
                normal=states.get("__NORMAL"),
                normal_active=states.get("__NORMAL_ACTIVE"),
                hilited=states.get("__HILITED"),
                hilited_active=states.get("__HILITED_ACTIVE"),
                clicked=states.get("__CLICKED"),
                clicked_active=states.get("__CLICKED_ACTIVE"),
                normal_sticky=states.get("__NORMAL_STICKY"),
                normal_active_sticky=states.get("__NORMAL_ACTIVE_STICKY"),
            )
        return typed, raw
    ```

    Step C: `src/themey/analyze/tclasses.py`:

    ```python
    """AST __TCLASS blocks → TClassSpec dict.

    Tolerates the misspelled `__FORGROUND_COLOR` (E16's primary form) plus
    `__FOREGROUND_COLOR` (correct spelling) and `__COLOR` (alternate alias).
    """
    from __future__ import annotations

    from themey.etheme.ast import Block, KeyVal
    from themey.ir import TClassSpec

    FG_COLOR_KEYS: tuple[str, ...] = (
        "__FORGROUND_COLOR",     # E16's misspelling — primary
        "__FOREGROUND_COLOR",    # correct spelling — fallback
        "__COLOR",               # alternate alias
    )

    TCLASS_STATE_CONTEXT_KEYS: frozenset[str] = frozenset({
        "__NORMAL", "__NORMAL_ACTIVE",
        "__HILITED", "__HILITED_ACTIVE",
        "__CLICKED", "__CLICKED_ACTIVE",
        "__NORMAL_STICKY", "__NORMAL_ACTIVE_STICKY",
    })


    def build_tclasses(tclass_blocks: list[Block]) -> dict[str, TClassSpec]:
        out: dict[str, TClassSpec] = {}
        for block in tclass_blocks:
            if not block.head_values:
                continue
            name = str(block.head_values[0])
            current_state: str | None = None
            colors: dict[str, tuple[int, int, int]] = {}
            for kv in block.children:
                if not isinstance(kv, KeyVal):
                    continue
                if kv.keyword in TCLASS_STATE_CONTEXT_KEYS and not kv.values:
                    current_state = kv.keyword
                elif kv.keyword in FG_COLOR_KEYS and len(kv.values) >= 3:
                    if current_state is not None:
                        try:
                            rgb = (int(kv.values[0]), int(kv.values[1]), int(kv.values[2]))
                        except (ValueError, TypeError):
                            continue
                        colors[current_state] = rgb
            out[name] = TClassSpec(
                name=name,
                fg_normal=colors.get("__NORMAL"),
                fg_active=colors.get("__NORMAL_ACTIVE"),
            )
        return out
    ```
